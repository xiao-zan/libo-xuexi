pwd      查看当前位置
ls       查看当前目录
ls -la   查看隐藏目录
cd       进入指定目录
cd ..    进入上一级目录
mkdir    创建文件夹
touch    创建文件
rm       删除文件
rmdir    删除空文件
cat      查看文件内容
clear    清除终端上下文
./       当前目录
/        根目录
~        家目录

    shift + :
    wq （保存+退出）或 q!（只退出不保存）
}
